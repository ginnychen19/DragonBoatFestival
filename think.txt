main.js
01. init 
會包含一個class一切的初始化，包含一個 update 和 draw
     01.update
             a. 時間計數器 =>用deltaTime的時間去累加在轉換格式
             b. 遊戲是否結束判斷 if (this.time > this.maxTime) this.gameOver = true;
             c. 遊戲背景、船隻資訊的更新
                    this.background.update();
                    this.player.update(this.input.keys, deltaTime);
             d. 生成障礙物
      02.draw 就是處理他們的draw行為
            this.background.draw(context);
            this.player.draw(context);
            this.enemies.forEach(enemy => {
                enemy.draw(context);
            })
            this.UI.draw(context)

02. animate
        因為要遵循update-collide-render週期，所以
            //要加clearRect，要不然會有無限殘影。
               ctx.clearRect(0, 0, canvas.width, canvas.height);         
               game.update(deltaTime);
               game.draw(ctx); //會把ctx傳進各個js裡
            //不是this.gameOver喔！
               if (!game.gameOver) requestAnimationFrame(animate);
